To get the application URL, run these commands:
{{- $ns := .Release.Namespace }}
{{- $bl := printf "%s-bl" (include "bridgelink.fullname" .) }}
{{- $svc := .Values.bridgelink.service }}
{{- /* AWS reports a load balancer hostname and most other providers an IP, so print whichever is set. */}}
{{- $lbAddr := "{.status.loadBalancer.ingress[0].hostname}{.status.loadBalancer.ingress[0].ip}" }}

{{- if eq $svc.type "LoadBalancer" }}
  NOTE: It may take a few minutes for the load balancer address to be available.
        Watch the status with: 'kubectl get svc --namespace {{ $ns }} -w {{ $bl }}'

  export SERVICE_HOST=$(kubectl get svc --namespace {{ $ns }} {{ $bl }} -o jsonpath='{{ $lbAddr }}')
  echo "BridgeLink URLs:"
  {{- if $svc.ports.http }}
  echo "- HTTP:  http://$SERVICE_HOST:{{ $svc.ports.http }}"
  {{- end }}
  echo "- HTTPS: https://$SERVICE_HOST:{{ $svc.ports.https }}"

{{- else if eq $svc.type "NodePort" }}
  export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[0].address}')
  export NODE_PORT=$(kubectl get svc --namespace {{ $ns }} {{ $bl }} -o jsonpath='{.spec.ports[?(@.name=="https")].nodePort}')
  echo "BridgeLink HTTPS: https://$NODE_IP:$NODE_PORT"

{{- else }}
  To access BridgeLink from within the cluster:
  echo "BridgeLink URLs:"
  {{- if $svc.ports.http }}
  echo "- HTTP:  http://{{ $bl }}.{{ $ns }}.svc:{{ $svc.ports.http }}"
  {{- end }}
  echo "- HTTPS: https://{{ $bl }}.{{ $ns }}.svc:{{ $svc.ports.https }}"

  To access from outside the cluster, run these commands in a shell:
  {{- if $svc.ports.http }}
  kubectl port-forward --namespace {{ $ns }} svc/{{ $bl }} {{ $svc.ports.http }}:{{ $svc.ports.http }} &
  {{- end }}
  kubectl port-forward --namespace {{ $ns }} svc/{{ $bl }} {{ $svc.ports.https }}:{{ $svc.ports.https }} &

  Then access:
  {{- if $svc.ports.http }}
  - HTTP:  http://127.0.0.1:{{ $svc.ports.http }}
  {{- end }}
  - HTTPS: https://127.0.0.1:{{ $svc.ports.https }}
{{- end }}

{{- if and (not .Values.bridgelink.persistence.enabled) (not .Values.bridgelink.keystore.existingSecret) }}

WARNING: appdata is an emptyDir and no keystore Secret is set. Whenever the BridgeLink pod is
  replaced (node drain, upgrade, rescheduling) it starts with a new data-encryption key, and message
  content stored with encryption on before then can no longer be read. Set
  bridgelink.persistence.enabled=true or bridgelink.keystore.existingSecret.
{{- else }}

Back up appdata/keystore.jks and its passwords: they hold the data-encryption key and TLS
certificate, and encrypted message content cannot be read without them. See the chart README.
{{- end }}
